**for windows install git bash
1. Go to your project
cd ~/personal-knowledge-base
git remote -v
git status

2. Create a new file
touch ms-pool_platform_worker.sql
ls
3. Open the file and add content
code ms-pool_platform_worker.sql

4. Check what Git sees
git status

5. Add the file to Git
git add ./
git status

6. Create a commit
git commit -m "Add platform worker SQL"

7. Upload/push to GitHub
gh auth setup-git
git push origin main
7.1 if not available for authentication type (for windowss)
  winget install --id GitHub.cli(close and reopen bash)
  gh --version
  gh auth login
  gh auth setup-git
  gh auth status
  git status
7.2 git identity using global username or email
    git config --global user.name "Prashant"
    git config --global user.email "prashant.k2411@gmail.com"
    git commit -m "Add platform worker SQL" 


8. Check that everything is synchronized
git status
